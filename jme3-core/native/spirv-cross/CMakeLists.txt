cmake_minimum_required(VERSION 3.20)
project(jme_spirv_cross LANGUAGES CXX)
find_package(JNI REQUIRED)

if(NOT EXISTS "${SPIRV_CROSS_SOURCE_DIR}/spirv_cross_c.h")
    message(FATAL_ERROR "Set SPIRV_CROSS_SOURCE_DIR to the pinned SPIRV-Cross checkout")
endif()

set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(MSVC)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
set(SPIRV_CROSS_CLI OFF)
set(SPIRV_CROSS_ENABLE_TESTS OFF)
set(SPIRV_CROSS_STATIC ON)
set(SPIRV_CROSS_SHARED OFF)
set(SPIRV_CROSS_ENABLE_GLSL ON)
set(SPIRV_CROSS_ENABLE_HLSL OFF)
set(SPIRV_CROSS_ENABLE_MSL OFF)
set(SPIRV_CROSS_ENABLE_CPP OFF)
set(SPIRV_CROSS_ENABLE_REFLECT OFF)
set(SPIRV_CROSS_ENABLE_UTIL OFF)
set(SPIRV_CROSS_ENABLE_C_API ON)
set(SPIRV_CROSS_SKIP_INSTALL ON)
add_subdirectory("${SPIRV_CROSS_SOURCE_DIR}" upstream EXCLUDE_FROM_ALL)

add_library(jme-spirv-cross SHARED SpirvCrossBridge.cpp)
target_compile_features(jme-spirv-cross PRIVATE cxx_std_11)
target_include_directories(jme-spirv-cross PRIVATE ${JNI_INCLUDE_DIRS})
target_link_libraries(jme-spirv-cross PRIVATE spirv-cross-c)
set_target_properties(jme-spirv-cross PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux" AND CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
    target_link_options(jme-spirv-cross PRIVATE -static-libgcc -static-libstdc++)
endif()

install(TARGETS jme-spirv-cross RUNTIME DESTINATION . LIBRARY DESTINATION .)
install(FILES "${SPIRV_CROSS_SOURCE_DIR}/LICENSE" DESTINATION . RENAME LICENSE-spirv-cross.txt)
