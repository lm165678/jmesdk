cmake_minimum_required(VERSION 3.20)
project(jme_xatlas LANGUAGES CXX)
find_package(JNI REQUIRED)
find_package(Threads REQUIRED)
find_file(XATLAS_CPP NAMES xatlas.cpp PATHS "${XATLAS_SOURCE_DIR}/source/xatlas" "${XATLAS_SOURCE_DIR}" NO_DEFAULT_PATH REQUIRED)
get_filename_component(XATLAS_INCLUDE "${XATLAS_CPP}" DIRECTORY)
add_library(jme-xatlas SHARED XatlasBridge.cpp "${XATLAS_CPP}")
target_compile_features(jme-xatlas PRIVATE cxx_std_11)
target_include_directories(jme-xatlas PRIVATE ${JNI_INCLUDE_DIRS} "${XATLAS_INCLUDE}")
target_link_libraries(jme-xatlas PRIVATE Threads::Threads)
install(TARGETS jme-xatlas RUNTIME DESTINATION . LIBRARY DESTINATION .)
install(FILES "${XATLAS_SOURCE_DIR}/LICENSE" DESTINATION . RENAME LICENSE-xatlas.txt)
