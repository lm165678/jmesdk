cmake_minimum_required(VERSION 3.24)
project(JmeTextureCodec LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
find_package(JNI REQUIRED COMPONENTS JVM)
find_package(Threads REQUIRED)

set(ASTCENC_CLI OFF CACHE BOOL "" FORCE)
set(ASTCENC_WERROR OFF CACHE BOOL "" FORCE)
set(ASTCENC_UNIVERSAL_BUILD OFF CACHE BOOL "" FORCE)
set(ASTCENC_ISA_NONE ON CACHE BOOL "" FORCE)
add_subdirectory("${ASTC_SOURCE}" astc EXCLUDE_FROM_ALL)

function(add_bc_codecs)
    # The upstream codec subdirectories locate shared sources using this variable.
    file(TO_CMAKE_PATH "${CMP_SOURCE}" PROJECT_SOURCE_DIR)
    set(PROJECT_FOLDER_SDK_LIBS "Compressonator")
    add_compile_definitions(USE_ETCPACK ENABLE_MAKE_COMPATIBLE_API USE_OLD_SWIZZLE)
    if(WIN32)
        add_compile_definitions(_CMP_CPP17_=1)
    else()
        add_compile_definitions(_CMP_CPP14_=1 _LINUX)
    endif()
    add_subdirectory("${CMP_SOURCE}/cmp_core" cmp-core EXCLUDE_FROM_ALL)
    add_subdirectory("${CMP_SOURCE}/cmp_compressonatorlib" cmp-codec EXCLUDE_FROM_ALL)
endfunction()
add_bc_codecs()

add_library(jme-texture-codec SHARED TextureCodec.cpp)
target_include_directories(jme-texture-codec PRIVATE ${JNI_INCLUDE_DIRS})
target_link_libraries(jme-texture-codec PRIVATE CMP_Compressonator astcenc-none-static Threads::Threads)
install(TARGETS jme-texture-codec RUNTIME DESTINATION . LIBRARY DESTINATION .)
