cmake_minimum_required(VERSION 3.18)
project(jme_ufbx_jni LANGUAGES C CXX)
set(UFBX_SOURCE_DIR "" CACHE PATH "Directory containing ufbx.c and ufbx.h")
if(WIN32)
    set(JNI_PLATFORM win32)
elseif(APPLE)
    set(JNI_PLATFORM darwin)
else()
    set(JNI_PLATFORM linux)
endif()
if(NOT EXISTS "$ENV{JAVA_HOME}/include/jni.h"
        OR NOT EXISTS "$ENV{JAVA_HOME}/include/${JNI_PLATFORM}/jni_md.h")
    message(FATAL_ERROR "Set JAVA_HOME to a JDK with headers for the build OS")
endif()
add_library(jme-ufbx SHARED UfbxBridge.cpp "${UFBX_SOURCE_DIR}/ufbx.c")
target_include_directories(jme-ufbx PRIVATE "${UFBX_SOURCE_DIR}"
    "$ENV{JAVA_HOME}/include" "$ENV{JAVA_HOME}/include/${JNI_PLATFORM}")
target_compile_features(jme-ufbx PRIVATE c_std_99 cxx_std_17)
if(MSVC)
    set_property(TARGET jme-ufbx PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
install(TARGETS jme-ufbx RUNTIME DESTINATION . LIBRARY DESTINATION .)
install(FILES "${UFBX_SOURCE_DIR}/LICENSE" DESTINATION . RENAME LICENSE-ufbx.txt)
